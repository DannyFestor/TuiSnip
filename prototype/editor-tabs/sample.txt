// Tab-aligned columns, the way people paste them (F3 flips tab stops and fixed width):
// name	kind	size
// a	file	12
// longer-name	dir	4096
// 日本語	wide	3
// 🙂	emoji	1

const (
	Short	= 1	// one tab after a short name
	MuchLongerName	= 2	// one tab after a long name
)

/*
	A block comment spanning lines: F7 "line" mode can't know these lines are a comment.
*/

	if r.Method != http.MethodGet {	// a tab before this comment
		http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
		return
	}
	for i := 0; i < 3; i++ {
		fmt.Fprintf(w, "%d\t%s\n", i, `a raw string
	with a tab inside it`)
	}

		// two tabs
}
